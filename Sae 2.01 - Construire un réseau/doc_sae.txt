<-----------------------------------------pour config du spanning tree --------------------------------------------------->
comm_fed (switch racine)
	spanning-tree mode pvst   #Per-VLAN Spanning Tree (1 instance STP par VLAN)
	spanning-tree extend system-id  #Permet à plusieurs instances STP (une par VLAN) de coexister sans conflit d’identifiants
	spanning-tree vlan 100 priority 4096  #les priorité doivent etre multiple de 4096
	spanning-tree vlan 200 priority 8192  #priorité la plus basse = racine
	spanning-tree vlan 300 priority 12288  #priorité par défaut = 32768
	
comm_1 
	spanning-tree mode pvst
	spanning-tree extend system-id
	spanning-tree vlan 100,200,300 priority 28672

<---------------------------------------------------------------------ACL sur R2 ----------------------------------------------------->
les ACL (access-list) permettent d'autoriser ou interdire du traffic
#le numéro doit etre entre 100 et 199 pour filtrer des protoccoles et ports specifiques
	access-list 106 permit tcp any any eq 80  #autorise tcp sur port 80 (http)
	access-list 106 permit tcp any any eq 22  #autorise tcp sur port 22 (ssh)
	access-list 106 permit tcp any any eq 873 #autorise tcp sur port 873 (rsynch pour le serveur web)
	access-list 106 deny ip any any			  #interdit tous les autres paquets ip (pas obligatoire car automatique)
	
	ensuite il faut appliquer cette liste a l'int voulue 
	int f0/0
		ip access-group 106 in #(traffic entrant)
	int f1/0
		ip access-group 106 out  #(traffic sortant)
	int f0/1
		ip access-group 106 in
