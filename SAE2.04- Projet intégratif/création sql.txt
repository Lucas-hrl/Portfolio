create database pj_integratif;
use pj_integratif;


CREATE TABLE capteurs (
  id_capteur VARCHAR(30) PRIMARY KEY,
  nom_capteur VARCHAR(100) UNIQUE NOT NULL,
  piece VARCHAR(50) NOT NULL,
  emplacement VARCHAR(100),
);
 
CREATE TABLE mesures (
  id_mesure INT AUTO_INCREMENT PRIMARY KEY,
  id_capteur VARCHAR(30),
  date_heure DATETIME NOT NULL,
  temperature DECIMAL(5,2),
  FOREIGN KEY (id_capteur) REFERENCES capteurs(id_capteur)
  	ON DELETE CASCADE
);
